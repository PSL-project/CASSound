cmake_minimum_required(VERSION 3.25)
project(CASSoundVST3 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CASSound VST3 requires x64")
endif()
set(VST3_SDK_ROOT "" CACHE PATH "VST3 SDK 3.8 checkout (MIT)")
set(JSON_ROOT "" CACHE PATH "nlohmann/json 3.12.0 checkout")
if(NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt" OR NOT EXISTS "${JSON_ROOT}/single_include/nlohmann/json.hpp")
  message(FATAL_ERROR "Set VST3_SDK_ROOT and JSON_ROOT; see README.md for pinned sources")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk)
if(WIN32)
  set(PLATFORM_MODULE "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp")
elseif(UNIX AND NOT APPLE)
  set(PLATFORM_MODULE "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_linux.cpp")
else()
  message(FATAL_ERROR "Supported host targets: Windows x64 and Linux verification")
endif()
add_executable(cassound-vst3 host.cpp ${PLATFORM_MODULE} "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp")
target_include_directories(cassound-vst3 PRIVATE "${VST3_SDK_ROOT}" "${JSON_ROOT}/single_include")
target_link_libraries(cassound-vst3 PRIVATE sdk_hosting ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(cassound-vst3 PRIVATE ole32 shell32)
endif()
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
smtg_add_vst3plugin(cassound-testfx fixture.cpp "${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
target_link_libraries(cassound-testfx PRIVATE sdk)
option(CASSOUND_BUILD_MDA_TESTS "Build official SDK mda plugins for additional compatibility tests" OFF)
if(CASSOUND_BUILD_MDA_TESTS)
  function(smtg_target_setup_as_vst3_example target)
    smtg_target_configure_version_file(${target})
    target_compile_features(${target} PUBLIC cxx_std_17)
    target_link_libraries(${target} PRIVATE sdk)
  endfunction()
  set(vstsdk_VERSION 3.8.1)
  set(SDK_ROOT "${VST3_SDK_ROOT}")
  add_subdirectory("${VST3_SDK_ROOT}/public.sdk/samples/vst/mda-vst3" mda)
endif()
